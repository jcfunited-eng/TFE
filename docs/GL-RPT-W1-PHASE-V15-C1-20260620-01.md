## F.1 — Five-channel `state_fivers` table

All values use existing `MODALITY_COMPONENTS` from `sensory_corpus.py`:
- **sight:** `{hue, saturation, lightness, edge_density, motion, shape_signature}` — all floats [0,1]
- **sound:** `{spectral_centroid, spectral_bandwidth, attack, decay, harmonicity, loudness}` — all floats [0,1]
- **touch:** `{pressure, temperature, vibration, texture_freq, sharpness, wetness}` — all floats [0,1]
- **smell:** `{sweet, putrid, floral, fruity, smoky, earthy, sour, fresh}` — all floats [0,1]
- **taste:** `{sweet, sour, bitter, salty, umami, intensity}` — all floats [0,1]

Zero-token convention: `{component: 0.0 for component in MODALITY_COMPONENTS[modality]}`. Represents silence/darkness/odorless/tasteless/no-contact. Never `None`, never `{}`.

## F.3 — `compute_room_sum` function

```python
# dsf_ai_service/substrate/world.py

from dsf_ai_service.sensory_corpus import MODALITY_COMPONENTS

ZERO_FIVER = {
    mod: {comp: 0.0 for comp in comps}
    for mod, comps in MODALITY_COMPONENTS.items()
}

def compute_room_sum(world_state, clock, object_states, object_defs):
    """Sum all five-channel signatures in the current place.

    Returns: dict of {modality: {component: float}} — the room's
    total sensory signature for this tick.

    Contributors:
    - Each object in the current place contributes its current state's fiver.
    - Container-gated objects (music_box, bell) contribute ONLY when
      their container is open.
    - Sky-items contribute through the window ONLY when drapes are open
      AND the sky-item is above the horizon.
    - Place baseline (ambient) is additive (W3-ready; W1 baseline is zero).
    """
    room = {
        mod: {comp: 0.0 for comp in comps}
        for mod, comps in MODALITY_COMPONENTS.items()
    }

    place = world_state.places[world_state.current_place]

    # Sum object fivers
    for obj_name in place.objects:
        obj_def = object_defs[obj_name]
        obj_state = object_states[obj_name]

        # Container gating: if object is inside a closed container, skip
        if obj_def.container:
            container_state = object_states[obj_def.container]
            if container_state.state != "open":
                continue

        # Get the fiver for the current state
        if obj_def.stateless:
            fiver = obj_def.state_fivers.get("always", ZERO_FIVER)
        else:
            fiver = obj_def.state_fivers.get(obj_state.state, ZERO_FIVER)

        # Add to room sum
        for mod in MODALITY_COMPONENTS:
            for comp in MODALITY_COMPONENTS[mod]:
                room[mod][comp] += fiver[mod][comp]

    # Sky-item contributions (through window, gated by drapes)
    drapes_state = object_states["drapes"].state
    if drapes_state == "open":
        if clock["sun_altitude"] > 0:
            sun_fiver = SKY_FIVERS["sun_sky"]
            for mod in MODALITY_COMPONENTS:
                for comp in MODALITY_COMPONENTS[mod]:
                    room[mod][comp] += sun_fiver[mod][comp]
        if clock["moon_altitude"] > 0:  # no is_day gate (Part J fix)
            moon_fiver = SKY_FIVERS["moon_sky"]
            for mod in MODALITY_COMPONENTS:
                for comp in MODALITY_COMPONENTS[mod]:
                    room[mod][comp] += moon_fiver[mod][comp]

    # Place ambient baseline (W3-ready; W1 = zero)
    # room += place.ambient_fiver  # uncomment when W3 adds weather

    return room
```

Notes:
- Values are summed, not clamped. The krimelack oscillator handles large signals gracefully (its winding count just increases). Clamping would lose distinguishability between "1 bright object" and "3 bright objects."
- The room-sum is recomputed per tick from current state. No delta operators, no `light_level_shift` effects.
- `compute_light_level` (from V1 §6) is still derived from `sun_altitude` + `drapes_state` + `night_light_state` — the room-sum's sight.lightness channel is a separate signal path that feeds the krimelack, not the dashboard's light_level indicator. Both read the same underlying state but serve different purposes (krimelack binding vs human-readable status).

## F.4 — `world_act` endpoint handler

```python
# In dsf_ai_service/app.py

class WorldActRequest(BaseModel):
    target_id: str
    verb: str
    source: str  # "joe" | "wc"

@app.post("/api/v1/world/act")
async def world_act(req: WorldActRequest):
    if req.source not in ("joe", "wc"):
        raise HTTPException(403, "invalid source")

    client = _get_substrate_client()
    result = await client.call("world_act",
        target_id=req.target_id, verb=req.verb, source=req.source)

    if result.get("error"):
        code = result.get("status", 409)
        raise HTTPException(code, result["error"])
    return result


# In dsf_ai_service/substrate_runner.py

def handle_world_act(args):
    target_id = args["target_id"]
    verb = args["verb"]
    source = args["source"]

    # Presence check
    if not _guala.coordinator._presence.get(source, False):
        return {"error": f"source '{source}' not present", "status": 409}

    # Object existence check
    if target_id not in _guala.world.object_states:
        return {"error": f"unknown object '{target_id}'", "status": 404}

    obj_def = _guala.world.object_defs[target_id]
    obj_state = _guala.world.object_states[target_id]

    # Location check
    if obj_def.home_place != _guala.world.current_place:
        return {"error": f"'{target_id}' not in current place", "status": 409}

    # Verb existence check
    if verb not in obj_def.verbs:
        return {"error": f"verb '{verb}' not defined on '{target_id}'", "status": 400}

    verb_def = obj_def.verbs[verb]

    # Autonomy-only check
    if verb_def.autonomy_only:
        return {"error": f"verb '{verb}' on '{target_id}' is autonomy-only", "status": 403}

    # Container gating
    if obj_def.container:
        container_state = _guala.world.object_states[obj_def.container]
        if container_state.state != "open":
            return {"error": f"container '{obj_def.container}' is closed", "status": 409}

    # State precondition check
    if verb_def.from_states and obj_state.state not in verb_def.from_states:
        return {"error": f"'{target_id}' is '{obj_state.state}', "
                         f"verb '{verb}' requires {verb_def.from_states}", "status": 409}

    # Flip state
    state_before = obj_state.state
    if verb_def.to_state is not None:
        obj_state.state = verb_def.to_state
    obj_state.use_count += 1
    obj_state.last_used_tick = _guala.tick

    # Artifact creation (crayons.draw)
    artifact_id = None
    if verb_def.creates_artifact:
        artifact_id = _guala.world.create_artifact(
            artifact_type="drawing", source=source,
            tick=_guala.tick, place=_guala.world.current_place)

    # Log object_action event
    _guala._log_substrate_event("object_action",
        verb=verb, target_id=target_id,
        state_before=state_before,
        state_after=obj_state.state,
        source=source,
        place=_guala.world.current_place,
        artifact_id=artifact_id)

    # Tag next tick's binding window with source
    _guala._world_action_source = source

    return {
        "ok": True,
        "tick": _guala.tick,
        "target_id": target_id,
        "verb": verb,
        "state_before": state_before,
        "state_after": obj_state.state,
        "source": source,
        "artifact_id": artifact_id,
    }

# Add to DISPATCH_TABLE:
# "world_act": handle_world_act,
```

Notes:
- `_world_action_source` is set so the next autonomy tick's binding window inherits the source tag. Cleared after one tick.
- No `guala_give_experience` call. No bundle. The room-sum recomputes on the next tick with the object's new state fiver.

def compute_window_view(clock: dict) -> list[dict]:
