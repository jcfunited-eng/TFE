\section*{Abstract}

We present a conservative field framework --- the Universal Field of Coherent
Processes (UFCP) --- in which conventional ``numbers'' are reinterpreted as
processual quantities that carry local tempo, phase, and coherent potential.
From a single Hermitian action with symmetric couplings we derive equations
of motion, Noether currents, and precise correspondence limits that reproduce
(i)~the Schr\"odinger equation and Born's rule, (ii)~Newtonian mechanics via
Ehrenfest/collective-coordinate limits, and (iii)~relativistic wave dynamics
with microcausality in Klein--Gordon/Dirac corners.

Measurement probabilities are derived by extremizing captured coherent
potential under four structural constraints, uniquely yielding the
overlap-squared rule. The framework makes falsifiable predictions including
a coupling-induced tempo dilation of coherently ``crowded'' oscillators and
a coherence-resource trade-off in multi-path interferometry.

We outline laboratory protocols and list explicit failure conditions that
would falsify the approach. This unifies classical, quantum, and relativistic
behaviors as projections of a single process-based dynamics while remaining
testable in precision metrology.

\section{Introduction}

We develop a process-based field description where quantities are not static
scalars but coherent processes carrying a local tempo $\omega$, phase $\phi$,
and coherent-potential density $\rho$.

The ambition is pragmatic: recover known laws (classical, quantum,
relativistic) from one conservative action while exposing new, falsifiable
effects in precision metrology. Our construction is conservative in the
Noether sense (global $U(1)$, Poincar\'e, and gauge symmetries imply
conserved currents) and reduces to familiar theories in appropriate limits.

\subsection{Design Principles}

\begin{enumerate}[label=\arabic*.]
  \item \textbf{Single action.} All dynamics derive from one action
        functional $S$.
  \item \textbf{Hermitian operators.} The kinetic operator $\hat{\Omega}_g$
        is Hermitian, ensuring real eigenvalues and unitary evolution.
  \item \textbf{Symmetric interactions.} The interaction kernel $W$ is
        symmetric, ensuring norm conservation.
  \item \textbf{Falsifiability.} The framework makes specific, quantitative
        predictions testable with current equipment.
  \item \textbf{Correspondence.} Known physics must emerge as exact limits.
\end{enumerate}

\section{Coherence Field and Action}

\begin{definition}[Coherence Field]
Let $\psi = \sqrt{\rho}\, e^{i\phi}$ be the coherence field on a
Lorentzian manifold $(M, g_{\mu\nu})$ with minimal gauge coupling
$D_\mu = \partial_\mu + i\frac{q}{\hbar}A_\mu$.

The field carries three physical attributes:
\begin{itemize}[nosep]
  \item $\rho(\mathbf{x}, t) \geq 0$: coherent-potential density
  \item $\phi(\mathbf{x}, t) \in \mathbb{R}$: phase
  \item $\omega = \partial_t \phi$: local tempo
\end{itemize}
\end{definition}

\begin{axiom}[UFCP Action]
\begin{equation}
  \boxed{
  S = \int d^4x\, \sqrt{-g} \left[
    \frac{i\hbar}{2}\!\left(\psi^* D_t\psi - \psi D_t\psi^*\right)
    - \psi^* \hat{\Omega}_g \psi
    - U(\rho)
    - \frac{1}{2}\int \rho(\mathbf{x})\, W(\mathbf{x}-\mathbf{x}')\,
      \rho(\mathbf{x}')\, \sqrt{-g'}\, d^4x'
  \right]
  }
  \label{eq:action}
\end{equation}
where $\hat{\Omega}_g$ is a Hermitian kinetic operator, $U(\rho)$ is the
local self-interaction potential (bounded below), and
$W(\mathbf{x}-\mathbf{x}')$ is the symmetric interaction kernel.
\end{axiom}

\subsection{Equation of Motion}

Variation of $S$ with respect to $\psi^*$ yields:
\begin{equation}
  \boxed{
  i\hbar D_t \psi = \hat{\Omega}_g \psi
    + \frac{\partial U}{\partial \rho}\,\psi
    + (W * \rho)\,\psi
  }
  \label{eq:eom}
\end{equation}

\textbf{Verification:} The $U(\rho)$ term contributes
$\frac{\partial U}{\partial \rho}\psi$ because $\rho = |\psi|^2$ and
$\frac{\delta \rho}{\delta \psi^*} = \psi$. The $W$ term contributes
$(W * \rho)\psi$ by the same chain rule. Confirmed by direct computation.

\subsection{Conserved Currents}

Global $U(1)$ symmetry ($\psi \to e^{i\alpha}\psi$) gives:
\begin{equation}
  \nabla_\mu J^\mu = 0, \qquad
  J^0 = \rho, \qquad
  \mathbf{J} = \rho\, \mathbf{v}
  \label{eq:current}
\end{equation}
where $\mathbf{v} = \frac{\hbar}{m}\!\left(\nabla\phi
- \frac{q}{\hbar}\mathbf{A}\right)$ for quadratic $\hat{\Omega}_g$.

\section{Interaction Kernel Symmetry}

\begin{axiom}[Kernel Symmetry]
$W(\mathbf{x}-\mathbf{x}') = W(\mathbf{x}'-\mathbf{x})$.
\end{axiom}

\textbf{v2.0:} Grounded in CPT invariance.

\begin{proposition}[Symmetry from CPT]
If the dynamics are CPT-invariant, then the two-point density interaction
kernel must be symmetric under argument exchange.
\end{proposition}

\begin{proof}[Argument]
Under CPT, $\rho(\mathbf{x},t) \to \rho(-\mathbf{x},-t)$. The interaction
term transforms as:
\[
  \int \rho(\mathbf{x})\, W(\mathbf{x}-\mathbf{x}')\, \rho(\mathbf{x}')\, d^4x\, d^4x'
  \;\xrightarrow{\text{CPT}}\;
  \int \rho(\mathbf{y})\, W(\mathbf{y}'-\mathbf{y})\, \rho(\mathbf{y}')\, d^4y\, d^4y'
\]
(substituting $\mathbf{y} = -\mathbf{x}$). For CPT invariance of the action,
$W(\mathbf{y}'-\mathbf{y}) = W(\mathbf{y}-\mathbf{y}')$.
\end{proof}

\begin{remark}
The weak force violates P and C individually but preserves CPT. Weak effects
enter through chiral couplings in $\hat{\Omega}_g$, not through $W$. If
nature required an asymmetric $W$, the Hamiltonian would not be Hermitian,
unitarity would fail, and the framework would be falsified.
\end{remark}

\section{Correspondence Proofs}

\subsection{Schr\"odinger Equation}

For $\hat{\Omega}_g = -\frac{\hbar^2}{2m}\nabla^2 + V$ and the
non-interacting limit ($U = 0$, $W = 0$), Equation~\ref{eq:eom} reduces
exactly to the Schr\"odinger equation. The Madelung decomposition gives
the quantum potential:
\begin{equation}
  Q = -\frac{\hbar^2}{2m}\frac{\nabla^2\sqrt{\rho}}{\sqrt{\rho}}
\end{equation}

\subsection{Classical Limit (Newton/Ehrenfest)}

\begin{theorem}[Classical Limit]
When $L \gg \lambda_{dB} = h/(mv)$ and $|\nabla Q| \ll |\nabla V|$,
the packet center obeys $m\ddot{\mathbf{X}} = -\nabla V(\mathbf{X})$.
\end{theorem}

\textbf{v2.0:} The mechanism by which $|\nabla Q|$ becomes negligible:

\begin{proposition}[Decoherence Mechanism]
When $\psi$ interacts with environmental degrees of freedom through $W$,
off-diagonal elements of the reduced density matrix decay at rate:
\begin{equation}
  \gamma_{dec} \sim N_{env}\, \langle |W|^2 \rangle\, \frac{(\Delta x)^2}{\hbar^2}
  \label{eq:decoherence}
\end{equation}
\end{proposition}

For macroscopic objects ($N_{env} \sim 10^{23}$), $\gamma_{dec}$ is
enormous and $Q$ becomes negligible on timescales far shorter than
mechanical dynamics. The classical limit is a consequence of $W$, not
a separate postulate.

\subsection{Relativistic Limit}

With relativistic kinetic operators:
\begin{itemize}[nosep]
  \item Klein--Gordon:
    $\hat{\Omega}_{KG} = -\hbar^2 g^{\mu\nu}D_\mu D_\nu + m^2 c^2$

the framework yields microcausal dynamics with bounded group velocity.
These are exact, not approximate.

\section{Measurement: Born's Rule from Structural Extremum}

\textbf{v2.0:} Born's rule is \textit{derived from different axioms},
not derived from nothing. This is an axiom trade, not axiom elimination.

\textbf{v2.0 critical correction:} The v1.0 claim that Born's rule arises
from ``maximizing $\sum_k p_k |\alpha_k|^2$'' is \textbf{mathematically
incorrect}. That functional is linear in $p_k$; its maximum on the
probability simplex is a vertex (all weight on the largest $|\alpha_k|^2$),
not Born's rule. The extremization claim has been removed.

Born's rule follows from Gleason's theorem given structural constraints,
not from an optimization.

Standard QM postulates Born's rule directly: $p_k = |\langle \psi_k | \psi_S \rangle|^2$.

The UFCP replaces this with four structural constraints:

\begin{axiom}[Measurement Constraints]
\label{ax:measurement}
Let $\{\psi_k\}$ be orthonormal detector modes in a Hilbert space of
dimension $d \geq 3$, and $\alpha_k = \langle \psi_k | \psi_S \rangle$.
The detection probabilities $\{p_k\}$ satisfy:
\begin{enumerate}[label=(M\arabic*)]
  \item \textbf{Norm conservation:} $\sum_k p_k = 1$
  \item \textbf{Unitary invariance:} $p_k$ depends only on $|\alpha_k|^2$,
        not on global phases
  \item \textbf{Orthogonal additivity:} For any decomposition of the
        Hilbert space into orthogonal subspaces, probabilities assigned
        to the subspace equal the sum of probabilities of the modes
        within it
  \item \textbf{Non-contextuality:} $p_k$ for mode $\psi_k$ does not
        depend on which other modes are included in the detector basis
\end{enumerate}
\end{axiom}

\begin{theorem}[Born's Rule from Gleason]
\label{thm:born}
In a Hilbert space of dimension $d \geq 3$, the unique probability measure
satisfying (M1)--(M4) is:
\begin{equation}
  p_k = \frac{|\alpha_k|^2}{\sum_m |\alpha_m|^2}
\end{equation}
For a complete detector ($\sum_m |\alpha_m|^2 = 1$), this is the standard
Born rule $p_k = |\alpha_k|^2$.
\end{theorem}

\begin{proof}[Proof]
By Gleason's theorem (1957): in a Hilbert space of dimension $\geq 3$,
any probability measure $\mu$ on the lattice of closed subspaces that is
(i)~countably additive on orthogonal projections and (ii)~normalized
($\mu(\mathcal{H}) = 1$) has the form $\mu(P) = \mathrm{tr}(\rho\, P)$
for a unique density operator $\rho$.

Constraints (M1) and (M3) provide the additivity and normalization
required by Gleason. Constraint (M2) restricts to pure-state measures
$\rho = |\psi_S\rangle\langle\psi_S|$. Constraint (M4) (non-contextuality)
ensures the measure depends only on the projection, not the basis.

Applying Gleason's result:
$p_k = \mathrm{tr}(|\psi_S\rangle\langle\psi_S|\cdot|\psi_k\rangle\langle\psi_k|)
= |\langle\psi_k|\psi_S\rangle|^2 = |\alpha_k|^2$.

This is exact and unique. No extremization is involved.
\end{proof}

\begin{remark}[Axiom Trade Assessment]
Standard QM postulates Born's rule as: ``the probability of outcome $k$
is $|\alpha_k|^2$.'' The UFCP replaces this with four structural
constraints and invokes Gleason's theorem. The constraints are geometric
(additivity, invariance, non-contextuality) rather than probabilistic
(``the probability is...'').

Whether this constitutes explanatory progress: the UFCP does not explain
\textit{why} probabilities arise in a deterministic field theory. It
explains \textit{which} probability rule must hold given structural
assumptions about measurement. This is a narrower but mathematically
precise claim.

The dimension restriction ($d \geq 3$) is inherited from Gleason's
theorem. In $d = 2$ (single qubit), the constraints do not uniquely
determine Born's rule --- additional structure is needed. This is a
known limitation, not a flaw: all physical Hilbert spaces have $d \geq 3$.
\end{remark}

\begin{remark}[v1.0 Error Correction]
The v1.0 formulation included ``reciprocity (bilinear capture)'' as (M4)
and claimed Born's rule from extremizing $\sum_k p_k |\alpha_k|^2$.
Numerical verification demonstrates this is incorrect: the linear
functional $\sum_k p_k c_k$ on the simplex $\sum p_k = 1$, $p_k \geq 0$
is maximized at a vertex ($p_{k^*} = 1$ for $k^* = \arg\max c_k$), not at
$p_k \propto c_k$. The extremization claim was introduced during v1.0
manuscript preparation and is not part of the core UFCP physics. It has
been removed and replaced with the correct Gleason derivation.
\end{remark}

\section{Predictions}

\subsection{Coupling-Induced Tempo Dilation (CITD)}

In $N$ coherently locked oscillators, the leader's tempo red-shifts:
\begin{equation}
  \frac{\Delta f}{f} \approx -\alpha \frac{N\, g_B}{Q_{eff}}
  \label{eq:citd}
\end{equation}

\textbf{v2.0: Magnitude estimate.} For optical clock oscillators
($N = 4$, $g_B \approx 10^{-3}$, $Q_{eff} \approx 10^{15}$):
\begin{equation}
  \frac{\Delta f}{f} \approx -4 \times 10^{-18}
\end{equation}

Current optical clock precision is $\sim 10^{-18}$. The prediction is at
the edge of current capability. With $N = 16$: $\Delta f/f \sim 10^{-17}$,
well within reach.

Null control: one-way power taps (no bidirectional coupling) must show
zero shift.

\subsection{Coherence-Resource Trade-Off}

In three-mode interferometry, visibilities obey:
\begin{equation}
  V_{AB}^2 + V_{AC}^2 \leq V_0^2
\end{equation}

Distinguishable from decoherence: coherent coupling produces quadratic
trade-off; dissipative coupling produces exponential decay. The experiment
must verify the quadratic form.

The stress-energy tensor from the UFCP action:
\begin{equation}
  T_{\mu\nu} = \frac{\hbar^2}{4m}\left(
    D_\mu\psi^* D_\nu\psi + D_\nu\psi^* D_\mu\psi
    - g_{\mu\nu} D^\alpha\psi^* D_\alpha\psi

  + g_{\mu\nu}\left(U(\rho) + \tfrac{1}{2}(W*\rho)\rho\right)
  - \rho\, g_{\mu\nu}\, \frac{\partial U}{\partial \rho}
  \label{eq:Tmunu}
\end{equation}

This sources Einstein's equations $G_{\mu\nu} = 8\pi G\, T_{\mu\nu}$.

\subsection{Where UFCP Differs from Standard Semiclassical Gravity}

\begin{enumerate}[label=\arabic*.]
  \item The interaction kernel $W$ contributes to $T_{\mu\nu}$ and
        back-reacts on geometry. Standard QFT computes on a fixed background.
  \item The quantum potential $Q$ has spatial structure that contributes
        to curvature. Standard expectation values average this out.
  \item \textbf{Falsifiable prediction:} Gravitational decoherence rate
        depends on $W$ (density-density coupling), not just Newtonian
        potential. Testable via massive superposition experiments
        (Bose--Marletto--Vedral type).
\end{enumerate}

\begin{remark}[Scope Limitation]
The UFCP does not solve quantum gravity. It provides a semiclassical
framework. Full quantum gravity would require promoting $g_{\mu\nu}$
to a coherence field. This is an open problem, explicitly acknowledged.
\end{remark}

\section{Failure Conditions}

\begin{enumerate}[label=F\arabic*.]
  \item Norm non-conservation with Hermitian $\hat{\Omega}_g$ and
        symmetric $W$.
  \item Measurement outcomes violating Born's rule despite
        (M1)--(M4) being satisfied.
  \item CITD null result with tight controls ($N \geq 4$, bidirectional
        PLL, systematic shifts excluded).
  \item Visibility trade-off not quadratic under coherent coupling.
  \item Gravitational decoherence matching Newtonian prediction but not
        $W$-dependent UFCP prediction.
  \item Nature requiring non-Hermitian $\hat{\Omega}_g$ or asymmetric $W$.
\end{enumerate}
